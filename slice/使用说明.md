# 切片工具包使用说明

## 简介

切片工具包提供查找、过滤、映射、去重、交集/并集/差集和排序。公开 API 都是串行的。

## 快速开始

```go
package main

import (
    "fmt"
    "github.com/iwen-conf/utils-pkg/slice"
)

func main() {
    numbers := []int{1, 2, 3, 4, 5}
    fmt.Println(slice.Contains(numbers, 3))

    unique := slice.Unique([]string{"a", "b", "a"})
    fmt.Println(unique)

    common := slice.Intersection([]int{1, 2, 3}, []int{2, 3, 4})
    fmt.Println(common)

    evens := slice.Filter(numbers, func(n int) bool { return n%2 == 0 })
    fmt.Println(evens)

    sorted := slice.Sort([]float64{3.2, 1.7, 2.3})
    fmt.Println(sorted)
}
```
